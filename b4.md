TransactionParserTest
package com.securebank.parser;

import com.fasterxml.jackson.databind.ObjectMapper;
import org.junit.jupiter.api.Test;

import java.util.List;

import static org.junit.jupiter.api.Assertions.*;

class TransactionParserTest {

    private final TransactionParser parser =
            new TransactionParser(new ObjectMapper());

    /**
     * JSON null
     */
    @Test
    void shouldReturnEmptyListWhenJsonIsNull() {

        List<TransactionDTO> result =
                parser.parse(null);

        assertTrue(result.isEmpty());
    }

    /**
     * JSON rỗng
     */
    @Test
    void shouldReturnEmptyListWhenJsonIsBlank() {

        List<TransactionDTO> result =
                parser.parse("");

        assertTrue(result.isEmpty());
    }

    /**
     * JSON sai format
     */
    @Test
    void shouldReturnEmptyListWhenJsonMalformed() {

        String json = "{invalid json";

        List<TransactionDTO> result =
                parser.parse(json);

        assertTrue(result.isEmpty());
    }

    /**
     * amount âm
     */
    @Test
    void shouldSkipNegativeAmountTransaction() {

        String json = """
            [
              {
                "id":"TX01",
                "amount":-100,
                "status":"SUCCESS",
                "transactionDate":"2025-06-01T10:00:00"
              }
            ]
            """;

        List<TransactionDTO> result =
                parser.parse(json);

        assertEquals(0, result.size());
    }

    /**
     * JSON hợp lệ
     */
    @Test
    void shouldParseValidTransaction() {

        String json = """
            [
              {
                "id":"TX01",
                "amount":1000,
                "status":"SUCCESS",
                "transactionDate":"2025-06-01T10:00:00"
              }
            ]
            """;

        List<TransactionDTO> result =
                parser.parse(json);

        assertEquals(1, result.size());

        TransactionDTO dto = result.get(0);

        assertEquals("TX01", dto.id());
    }
}
LedgerBalanceCalculatorTest
package com.securebank.ledger;

import org.junit.jupiter.api.Test;
import org.junit.jupiter.api.extension.ExtendWith;

import org.mockito.InjectMocks;
import org.mockito.junit.jupiter.MockitoExtension;

import java.util.Arrays;
import java.util.Collections;

import static org.junit.jupiter.api.Assertions.*;

@ExtendWith(MockitoExtension.class)
class LedgerBalanceCalculatorTest {

    @InjectMocks
    private LedgerBalanceCalculator calculator;

    /**
     * List null
     */
    @Test
    void shouldReturnZeroWhenListIsNull() {

        double result =
                calculator.calculateBranchBalance(
                        null,
                        "HN",
                        true);

        assertEquals(0D, result);
    }

    /**
     * List rỗng
     */
    @Test
    void shouldReturnZeroWhenListIsEmpty() {

        double result =
                calculator.calculateBranchBalance(
                        Collections.emptyList(),
                        "HN",
                        true);

        assertEquals(0D, result);
    }

    /**
     * Chỉ cộng account ACTIVE
     */
    @Test
    void shouldCalculateOnlyActiveAccounts() {

        Account account1 =
                new Account(
                        "HN",
                        "ACTIVE",
                        1000
                );

        Account account2 =
                new Account(
                        "HN",
                        "INACTIVE",
                        2000
                );

        double result =
                calculator.calculateBranchBalance(
                        Arrays.asList(account1, account2),
                        "HN",
                        true);

        assertEquals(1000D, result);
    }

    /**
     * activeOnly=false
     */
    @Test
    void shouldCalculateAllValidAccounts() {

        Account account1 =
                new Account(
                        "HN",
                        "ACTIVE",
                        1000
                );

        Account account2 =
                new Account(
                        "HN",
                        "INACTIVE",
                        2000
                );

        double result =
                calculator.calculateBranchBalance(
                        Arrays.asList(account1, account2),
                        "HN",
                        false);

        assertEquals(3000D, result);
    }

    /**
     * Balance âm không được cộng
     */
    @Test
    void shouldIgnoreNegativeBalance() {

        Account account1 =
                new Account(
                        "HN",
                        "ACTIVE",
                        -100
                );

        double result =
                calculator.calculateBranchBalance(
                        Arrays.asList(account1),
                        "HN",
                        false);

        assertEquals(0D, result);
    }
}
