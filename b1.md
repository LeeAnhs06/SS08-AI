Khi tích hợp dữ liệu từ Core Banking cũ, các lỗi thường gặp gồm:

Lỗi cấu trúc JSON

=> Bỏ qua bản ghi và ghi log.

package com.securebank.parser;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import java.math.BigDecimal;
import java.time.LocalDateTime;
import java.util.ArrayList;
import java.util.List;

public class TransactionParser {

    private static final Logger log =
            LoggerFactory.getLogger(TransactionParser.class);

    private final ObjectMapper objectMapper;

    public TransactionParser(ObjectMapper objectMapper) {
        this.objectMapper = objectMapper;
    }

    public List<TransactionDTO> parse(String rawJson) {

        List<TransactionDTO> result = new ArrayList<>();

        if (rawJson == null || rawJson.isBlank()) {
            log.error("Input JSON is null or empty");
            return result;
        }

        try {

            JsonNode root = objectMapper.readTree(rawJson);

            if (!root.isArray()) {
                log.error("Root JSON is not an array");
                return result;
            }

            for (JsonNode node : root) {

                try {

                    String id = node.path("id").asText();

                    if (id == null || id.isBlank()) {
                        log.error("Invalid id");
                        continue;
                    }

                    JsonNode amountNode = node.get("amount");

                    if (amountNode == null) {
                        log.error("Missing amount for id {}", id);
                        continue;
                    }

                    BigDecimal amount;

                    try {
                        amount = new BigDecimal(amountNode.asText());
                    } catch (Exception e) {
                        log.error("Invalid amount for id {}", id);
                        continue;
                    }

                    if (amount.compareTo(BigDecimal.ZERO) <= 0) {
                        log.error("Amount must be > 0 for id {}", id);
                        continue;
                    }

                    String status =
                            node.path("status").asText();

                    if (!"SUCCESS".equals(status)
                            && !"FAILED".equals(status)) {

                        log.error(
                                "Invalid status {} for id {}",
                                status,
                                id
                        );
                        continue;
                    }

                    String dateText =
                            node.path("transactionDate").asText();

                    LocalDateTime localDateTime;

                    try {
                        localDateTime =
                                LocalDateTime.parse(dateText);
                    } catch (Exception e) {

                        log.error(
                                "Invalid date {} for id {}",
                                dateText,
                                id
                        );
                        continue;
                    }

                    result.add(
                            new TransactionDTO(
                                    id,
                                    amount,
                                    status,
                                    localDateTime
                            )
                    );

                } catch (Exception ex) {

                    log.error(
                            "Error processing transaction record",
                            ex
                    );
                }
            }

        } catch (Exception e) {

            log.error("Unable to parse JSON", e);
        }

        return result;
    }
}

        BigDecimal amount,
