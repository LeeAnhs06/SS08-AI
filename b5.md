package com.securebank.reconciliation;

import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;

import java.math.BigDecimal;
import java.time.ZoneOffset;
import java.util.Map;
import java.util.function.Function;
import java.util.stream.Collectors;
import java.util.List;

@Slf4j
@Service
public class ReconciliationService {

    public void reconcile(
            List<Transaction> coreTxns,
            List<Transaction> partnerTxns) {

        Map<String, Transaction> partnerMap =
                partnerTxns.stream()
                        .collect(Collectors.toMap(
                                Transaction::getId,
                                Function.identity()));

        for (Transaction coreTxn : coreTxns) {

            Transaction partnerTxn =
                    partnerMap.get(coreTxn.getId());

            if (partnerTxn == null) {

                log.error(
                        "Missing transaction: {}",
                        coreTxn.getId()
                );

                continue;
            }

            var coreUtc =
                    coreTxn.getTransactionTime()
                           .withZoneSameInstant(
                                   ZoneOffset.UTC);

            var partnerUtc =
                    partnerTxn.getTransactionTime()
                              .withZoneSameInstant(
                                      ZoneOffset.UTC);

            if (!coreTxn.getAmount()
                    .equals(partnerTxn.getAmount())) {

                log.error(
                        "Amount mismatch id={}",
                        coreTxn.getId()
                );

                throw new AmountMismatchException(
                        coreTxn.getId()
                );
            }

            if (!coreTxn.getStatus()
                    .equals(partnerTxn.getStatus())) {

                log.error(
                        "Status mismatch id={}",
                        coreTxn.getId()
                );

                throw new StatusMismatchException(
                        coreTxn.getId()
                );
            }

            log.info(
                    "Matched transaction id={} utc={}",
                    coreTxn.getId(),
                    coreUtc
            );
        }
    }
}

        extends RuntimeException {

    public AmountMismatchException(String id) {

        super(
            "Amount mismatch for transaction: " + id
        );
    }
}
